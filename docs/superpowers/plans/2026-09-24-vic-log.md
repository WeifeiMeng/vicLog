# Vic Log Personal Blog Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a statically generated Next.js personal blog whose site profile and posts are maintained as Markdown files.

**Architecture:** Next.js App Router pages call a server-only `lib/content` module that validates Markdown frontmatter and returns typed site and post objects. Reusable presentation components render the persistent author sidebar, article cards, and Markdown body; all public routes are statically exported, including RSS.

**Tech Stack:** Next.js 16.3.6, React 19.3.0, TypeScript 7.0.2, CSS Modules, gray-matter 4.0.3, react-markdown 10.1.0, remark-gfm 4.0.1, Vitest 5.0.1

---

```ts
import { afterEach, describe, expect, it } from "vitest";
import { mkdtempSync, mkdirSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import path from "node:path";
import { loadContent } from "./index";

const roots: string[] = [];
const fixture = (posts: Record<string, string>) => {
  const root = mkdtempSync(path.join(tmpdir(), "vic-log-"));
  roots.push(root);
  mkdirSync(path.join(root, "posts"));
  writeFileSync(path.join(root, "site.md"), "---\nname: Vic Log\nauthor: Meng Weifei\ntagline: ''\n---\nAbout body\n");
  for (const [name, body] of Object.entries(posts)) writeFileSync(path.join(root, "posts", name), body);
  return root;
};

afterEach(() => roots.splice(0).forEach((root) => rmSync(root, { recursive: true })));

describe("loadContent", () => {
  it("filters drafts and sorts published posts newest first", () => {
    const root = fixture({
      "older.md": "---\ntitle: Older\ndate: 2026-01-01\nexcerpt: Old\npublished: true\n---\nOld body",
      "newer.md": "---\ntitle: Newer\ndate: 2026-02-01\nexcerpt: New\npublished: true\n---\nNew body",
      "draft.md": "---\ntitle: Draft\ndate: 2026-03-01\nexcerpt: Draft\npublished: false\n---\nDraft body"
    });
    expect(loadContent(root).posts.map((post) => post.slug)).toEqual(["newer", "older"]);
  });

  it("reports the file and field for an invalid post", () => {
    const root = fixture({ "broken.md": "---\ntitle: Broken\ndate: no\nexcerpt: Text\npublished: true\n---\nBody" });
    expect(() => loadContent(root)).toThrow(/broken\.md.*date/);
  });
});
```
